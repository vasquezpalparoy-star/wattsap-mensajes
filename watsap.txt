    function initTheme() {
      const themeToggleBtn = document.getElementById('themeToggleBtn');

      const themeText = document.getElementById('themeText');

      const isDark = localStorage.getItem('theme') === 'dark' ||

    async function initFirebase() {
      try {
        const app = initializeApp(firebaseConfig);
        auth = getAuth(app);
        db = getFirestore(app);

        const userCred = await signInAnonymously(auth);
        currentUser = userCred.user;
        updateStatusUI(true);

        const templatesColRef = collection(db, 'artifacts', appId, 'public', 'data', 'templates');
        
        onSnapshot(templatesColRef, (snapshot) => {
          if (snapshot.empty) {
            seedTemplates.forEach(async (tpl) => {
              const docRef = doc(templatesColRef, tpl.id);
              await setDoc(docRef, tpl);
            });
          } else {
            templates = snapshot.docs.map(doc => ({ id: doc.id, ...doc.data() }));
            renderAll();
          }
        }, (error) => {
          console.error("Firestore error:", error);
          updateStatusUI(false);
        });

      } catch (err) {
        console.error("Firebase init error:", err);
        updateStatusUI(false);
        renderAll();
      }
    }

    function extractVariables(text) {
      if (!text) return [];
      // Supports both [Variable] and {Variable} syntax
      const squareMatches = text.match(/\[([^\]]+)\]/g) || [];
      const curlyMatches = text.match(/\{([^}]+)\}/g) || [];
      const combined = [...squareMatches, ...curlyMatches];
      return [...new Set(combined.map(m => m.replace(/[\[\]\{\}]/g, '').trim()))];
    }

    function formatCardBodyWithPills(text) {
      if (!text) return "";
      let formatted = text.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");

      formatted = formatted.replace(/\[([^\]]+)\]|\{([^}]+)\}/g, (match, p1, p2) => {
        const varName = p1 || p2;
        const lower = varName.toLowerCase();
        let colorClasses = "bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-950 dark:text-amber-300 dark:border-amber-800";
        
        if (lower.includes("fecha") || lower.includes("hora")) {
          colorClasses = "bg-emerald-100 text-emerald-800 border-emerald-200 dark:bg-emerald-950 dark:text-emerald-300 dark:border-emerald-800";
        } else if (lower.includes("precio") || lower.includes("monto") || lower.includes("cuenta")) {
          colorClasses = "bg-amber-100 text-amber-800 border-amber-200 dark:bg-amber-950 dark:text-amber-300 dark:border-amber-800";
        }

        return `<span class="inline-block px-1.5 py-0.5 my-0.5 rounded text-[11px] font-bold border ${colorClasses}">[${varName}]</span>`;
      });

      return formatted;
    }

    function openUseModal(templateId) {
      const tpl = templates.find(t => t.id === templateId);
      if (!tpl) return;

      selectedUseTemplate = tpl;
      useFieldValues = {};

      document.getElementById('useModalTitle').innerText = tpl.title;
      document.getElementById('useModalCategory').innerText = tpl.category;

      const vars = extractVariables(tpl.body);
      const container = document.getElementById('useDynamicFieldsContainer');

      if (vars.length === 0) {
        container.innerHTML = `
          <div class="p-3 bg-slate-50 dark:bg-slate-900 border border-slate-200 dark:border-slate-700 rounded-xl text-xs font-medium text-slate-600 dark:text-slate-400">
            Esta plantilla no contiene variables dinamicas. Puedes copiarla directamente.
          </div>
        `;
      } else {
        container.innerHTML = vars.map(v => `
          <div>
            <label class="block text-xs font-extrabold text-slate-800 dark:text-slate-200 mb-1">
              ${v}
            </label>
            <input 
              type="text" 
              data-usevar="${v}"
              placeholder="Escribir ${v}..."
              class="use-var-input w-full px-3.5 py-2 text-xs font-semibold border border-slate-300 dark:border-slate-600 rounded-xl focus:ring-2 focus:ring-[#008069] focus:outline-none bg-slate-50 dark:bg-slate-900 text-slate-900 dark:text-white"
            />
          </div>
        `).join('');

        container.querySelectorAll('.use-var-input').forEach(input => {
          input.addEventListener('input', (e) => {
            const varName = e.target.dataset.usevar;
            useFieldValues[varName] = e.target.value;
            updateUsePreview();
          });
        });
      }

      updateUsePreview();
      document.getElementById('useTemplateModal').classList.remove('hidden');
    }

    function closeUseModal() {
      document.getElementById('useTemplateModal').classList.add('hidden');
      selectedUseTemplate = null;
      useFieldValues = {};
    }

    function generateFormattedUseText() {
      if (!selectedUseTemplate) return "";
      let text = selectedUseTemplate.body;
      const vars = extractVariables(selectedUseTemplate.body);

      vars.forEach(v => {
        const val = useFieldValues[v] && useFieldValues[v].trim() !== "" ? useFieldValues[v] : `[${v}]`;
        text = text.replaceAll(`[${v}]`, val).replaceAll(`{${v}}`, val);
      });

      return text;
    }

    function updateUsePreview() {
      const previewEl = document.getElementById('usePreviewText');
      const text = generateFormattedUseText();
      previewEl.innerText = text || "Cargando mensaje...";

      const now = new Date();
      const hours = String(now.getHours()).padStart(2, '0');
      const mins = String(now.getMinutes()).padStart(2, '0');
      document.getElementById('usePreviewTime').innerText = `${hours}:${mins}`;
    }

    function copyTextToClipboard(text) {
      if (!text) return;
      const textarea = document.createElement('textarea');
      textarea.value = text;
      textarea.style.position = 'fixed';
      textarea.style.opacity = '0';
      document.body.appendChild(textarea);
      textarea.select();

      try {
        document.execCommand('copy');
        showToast("¡Mensaje copiado al portapapeles!");
      } catch (err) {
        showToast("Error al copiar texto.", "error");
      } finally {
        document.body.removeChild(textarea);
      }
    }

    function renderAll() {
      renderCategoryDropdown();
      renderTemplateGrid();
    }

    async function saveTemplate(tplData) {
      if (db && currentUser) {
        try {
          const docId = tplData.id || "tpl_" + Date.now();
          const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'templates', docId);
          await setDoc(docRef, { ...tplData, id: docId });
          showToast("Plantilla guardada exitosamente.");
        } catch (e) {
          console.error("Firestore save error:", e);
          fallbackSaveLocal(tplData);
        }
      } else {
        fallbackSaveLocal(tplData);
      }
    }

    function fallbackSaveLocal(tplData) {
      const idx = templates.findIndex(t => t.id === tplData.id);
      if (idx >= 0) {
        templates[idx] = tplData;
      } else {
        const newId = tplData.id || "tpl_" + Date.now();
        templates.push({ ...tplData, id: newId });
      }
      renderAll();
      showToast("Guardado en almacenamiento local.");
    }

    async function deleteTemplate(id) {
      if (db && currentUser) {
        try {
          const docRef = doc(db, 'artifacts', appId, 'public', 'data', 'templates', id);
          await deleteDoc(docRef);
          showToast("Plantilla eliminada.");
        } catch (e) {
          console.error("Firestore delete error:", e);
          fallbackDeleteLocal(id);
        }
      } else {
        fallbackDeleteLocal(id);
      }
    }

    function fallbackDeleteLocal(id) {
      templates = templates.filter(t => t.id !== id);
      renderAll();
      showToast("Plantilla eliminada.");
    }

    function showToast(msg, type = "success") {
      const toast = document.getElementById('toast');
      const toastMsg = document.getElementById('toastMsg');
      const toastIcon = document.getElementById('toastIcon');

      toastMsg.innerText = msg;
      toastIcon.className = type === "error" ? "fa-solid fa-circle-exclamation text-rose-500 text-lg" : "fa-solid fa-circle-check text-emerald-400 text-lg";

      toast.classList.remove('translate-y-24', 'opacity-0');
      setTimeout(() => {
        toast.classList.add('translate-y-24', 'opacity-0');
      }, 2800);
    }
